`GET notes/{id}` - возвращает в форматированном ввиде содержимое заметки, как мы сделали на уроке:
- 404 если заметку не нашли + текст о том что заметка не найдена
- 200 если нашли + форматированное содержимое заметки

`POST notes/{name}-{text}` - создает заметку, возвращает id заметки:
- 200 + id

`DELETE notes/{id}` - удаляет заметку:
- 200
### Как изменить хранилище под ID заметки?
``` go
type Note struct {
	NoteID int64
	Name string
	Text string
}

type Storage struct {
    hashmap map[int64]Note
    lastID  int64 // Всегда будет расти
}

func NewStorage() *Storage {
    return &Storage{
        hashmap: make(map[int64]Note),
        lastID:  0,
    }
}

func (s *Storage) Set(note Note) int64 {
    s.lastID++
    note.NoteID = s.lastID
    s.hashmap[note.NoteID] = note
    return note.NoteID
}

func (s *Storage) Get(id int64) (Note, error) {
    note, ok := s.hashmap[id]
    if !ok {
	    return Note{}, errors.New("note not found")
    }
    return note, ok
}

func (s *Storage) Remove(id int64) {
	delete(s.hashmap, id)
}
```
