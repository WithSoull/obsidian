![[99 - Meta/02 - Медиа/Pasted image 20240426004722.png]]

Смотрим есть ли в сете число на единицу больше:
1) если нет, то это край последовательности, остается найти ее начало и посчитать длину
2) если да, то этот элемент мы просчитаем потом, когда найдем крайний элемент для этой последовательности

P. S. на го решение ровно наоборот, если это начало последовательности - идти до конца
# Решение

``` python
class Solution:
    def longestConsecutive(self, nums: List[int]) -> int:
        hashset = set(nums)
        max_length = 0
        length = 0

        for num in nums:
            if (num + 1) not in hashset:
                i = num - 1
                length = 1
                while i in hashset:
                    length += 1
                    i -= 1
                max_length = max(length, max_length)

        return max_length
```

``` go 
func longestConsecutive(nums []int) int {
	set := make(map[int]struct{})
	for _, num := range nums {
		set[num] = struct{}{}
	}

	max_length := 0
	for num := range set {
		if _, ok := set[num-1]; ok {
			continue
		}
		// fmt.Println(num)

		ok := true
		end := num
		for ; ok; end++ {
			_, ok = set[end]
		}

		length := end - num - 1
    // fmt.Println(num, end)
		if max_length < length {
			max_length = length
		}
	}

	return max_length
}
```
