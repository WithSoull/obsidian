```python
class Solution:
    def isPalindrome(self, s: str) -> bool:
        formatedStr = ""
        for char in s:
            if char.isalnum(): formatedStr += char.lower()
        return formatedStr == formatedStr[::-1]
```

```python
class Solution(object):
    def __init__(self):
        self.alph = set("0123456789qwertyuiopasdfghjklzxcvbnm")

    def isPalindrome(self, s):
        s = s.lower()
        i = 0
        j = len(s) - 1
        while i < j:
            while (s[i] not in self.alph) and (i < j):
                i += 1
            while (s[j] not in self.alph) and (i < j):
                j -= 1

            if s[i] != s[j]:
                return False

            i += 1
            j -= 1

        return True

```

``` go 
func isPalindrome(s string) bool {
	left := 0
	right := len(s) - 1
	for _ = range s {
		for ; left < len(s); left++ {
			if unicode.IsDigit(rune(s[left])) || unicode.IsLetter(rune(s[left])) {
				break
			}
		}

		for ; right > -1; right-- {
			if unicode.IsDigit(rune(s[right])) || unicode.IsLetter(rune(s[right])) {
				break
			}
		}

		if left >= right {
			break
		}

		if unicode.ToLower(rune(s[right])) != unicode.ToLower(rune(s[left])) {
			return false
		}

		left += 1
		right -= 1
	}
	return true
}
```
