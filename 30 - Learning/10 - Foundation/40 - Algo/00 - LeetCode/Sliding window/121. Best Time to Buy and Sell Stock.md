func maxProfit(prices []int) (profit int) {
    minPriceId := 0

    for i := 0; i < len(prices) - 1; i++ {
        if prices[i] < prices[minPriceId] {
            minPriceId = i
        }

        profit = max(profit, prices[i + 1] - prices[minPriceId])
    }
    
    return
}


func max(a, b int) int {
    if a > b {
        return a
    } else {
        return b
    }
}
