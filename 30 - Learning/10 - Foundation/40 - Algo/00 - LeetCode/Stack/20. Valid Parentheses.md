``` python
class Solution:
    def isValid(self, s: str) -> bool:
        stack = []
        for b in s:
            if len(stack) > 0:
                a = stack[-1]
                if a == "(" and b == ")":
                    stack.pop()
                elif a == "{" and b == "}":
                    stack.pop()
                elif a == "[" and b == "]":
                    stack.pop()
                else:
                    stack.append(b)
            else:
                stack.append(b)

        return len(stack) == 0
```
