# Объяснение

1) Проверяем по строкам
2) Проверяем по столбцам
3) Проверяем боксы 3х3

![[99 - Meta/02 - Медиа/Pasted image 20240425235557.png]]

Если представим, что у боксов есть свои индексы по строкам и столбцам (от 0 до 2), тогда эти индексы можно найти, если поделить индексы самого элемента в матрице нацело на 3.
# Решение

``` python
class Solution:
    def isValidSudoku(self, board: List[List[str]]) -> bool:
        # check rows
        for row in board:
            s = set()
            for el in row:
                if el in s and el != ".":
                    return False
                s.add(el)

        # check columns
        for col in range(9):
            s = set()
            for row in range(9):
                el = board[row][col]
                if el in s and el != ".":
                    return False
                s.add(el)

        boxes = defaultdict(set)

        # check boxes
        for row in range(9):
            for col in range(9):
                el = board[row][col]
                if el in boxes[(row // 3, col // 3)] and el != ".":
                    return False
                boxes[(row // 3, col // 3)].add(el)

        return True
```

``` go 
func isValidSudoku(board [][]byte) bool {
	boxes := make(map[int]map[int]struct{})
	for i := 1; i < 4; i++ {
		for j := 1; j < 4; j++ {
			id := i*10 + j
			boxes[id] = make(map[int]struct{})
		}
	}
	for i := 0; i < 9; i++ {
		for j := 0; j < 9; j++ {
			if board[i][j] == '.' {
				continue
			}
			num := int(board[i][j])

			id := (i/3+1)*10 + (j/3 + 1)
			if _, ok := boxes[id][num]; ok {
				return false
			} else {
				boxes[id][num] = struct{}{}
			}
		}
	}

	for i := 0; i < 9; i++ {
		rows := make(map[int]struct{})
		for j := 0; j < 9; j++ {
			if board[i][j] == '.' {
				continue
			}
			num := int(board[i][j])

			if _, ok := rows[num]; ok {
				return false
			} else {
				rows[num] = struct{}{}
			}
		}
	}

	for j := 0; j < 9; j++ {
		cols := make(map[int]struct{})
		for i := 0; i < 9; i++ {
			if board[i][j] == '.' {
				continue
			}
			num := int(board[i][j])

			if _, ok := cols[num]; ok {
				return false
			} else {
				cols[num] = struct{}{}
			}
		}
	}

	return true
}
```

## Второе решение

Здесь вместо множеств используют булевы массивы, по времени такое решение выигрывает с маленьким отрывом, но вот по памяти оно явно лучше + намного чище код и в нем проще разобраться 
``` go 
func isValidSudoku(board [][]byte) bool {
	var rows [9][9]bool
	var cols [9][9]bool
	var squares [9][9]bool

	for r := 0; r < 9; r++ {
		for c := 0; c < 9; c++ {
        val := board[r][c]

			if val == '.' {
				continue
			}

			num := val - '1'

			squareIndex := (r/3)*3 + (c / 3)

			if rows[r][num] || cols[c][num] || squares[squareIndex][num] {
				return false
			}

			rows[r][num] = true
			cols[c][num] = true
			squares[squareIndex][num] = true
		}
	}

	return true
}
```
