# Объяснение

### Алгоритмическая сложность - O(n)

1) Создаем словарь Counter()
2) Создаем массив, где индекс элемента - это количество повторений, элемент это список этих элементов
3) В обратном порядке приходимся по этому массиву и берем K самых частых решения

Можно было создать транспонированный словарь для Counter() и отсортировать его по убыванию, затем вывести K элементов. Это решение хуже как минимум потому что сложность уже $O(n log(n))$, при этом сортировка в хэшмапах на некоторых языках - геморой, на джаве нужно самому писать qsort для этого. 

# Решение

```python
class Solution:
    def topKFrequent(self, nums: List[int], k: int) -> List[int]:
        cnt = Counter(nums)
        
        freq = [[] for _ in range(len(nums) + 1)]
        
        for num, count in cnt.items():
            freq[count].append(num)

        count = 0
        result = []
        for i in range(len(nums), 0, -1):
            for num in freq[i]:
                result.append(num)
                count += 1
                if count == k:
                    return result
```


``` go 
func topKFrequent(nums []int, k int) []int {
	hashmap := make(map[int]int)
	for _, num := range nums {
		hashmap[num]++
	}

	arr := make([][]int, len(nums)+1)

	for num, frequency := range hashmap {
		index := len(arr) - frequency
		arr[index] = append(arr[index], num)
	}
	answer := make([]int, 0)
	for _, list := range arr {
		for _, num := range list {
			answer = append(answer, num)
			if len(answer) == k {
				return answer
			}
		}
	}

	return nil
}
```

``` go 
func topKFrequent(nums []int, k int) []int {
	hashmap := make(map[int]int)
	for _, num := range nums {
		hashmap[num]++
	}

	numset := make([]int, 0)

	for num := range hashmap {
		numset = append(numset, num)
	}

	sort.Slice(numset, func(i, j int) bool { return hashmap[numset[i]] < hashmap[numset[j]] })

	return numset[len(numset)-k:]
}
```
