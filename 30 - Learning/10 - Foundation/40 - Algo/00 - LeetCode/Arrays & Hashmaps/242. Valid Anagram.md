# Объяснение

*Анаграммы* - слова, с одинаковым набором букв, включая точное количество каждой из них

# Решение

```python
class Solution:
    def isAnagram(self, s: str, t: str) -> bool:
        if len(s) != len(t):
            return False
        ds = {}
        dt = {}

        for i in range(len(t)):
            ds[s[i]] = 1 + ds.get(s[i], 0)
            dt[t[i]] = 1 + dt.get(t[i], 0)

        return ds == dt
```

``` go 
func isAnagram(s string, t string) bool {
    if len(s) != len(t) {
        return false
    }

    sMap := make(map[byte]int)
    tMap := make(map[byte]int)

    for i := 0; i < len(s); i++ {
        sMap[s[i]]++
        tMap[t[i]]++
    }

    for key, val := range sMap {
        if _, ok := tMap[key]; !ok {
            return false
        }
        if tMap[key] != val {
            return false
        }
    }
    return true
}
```

func isAnagram(s string, t string) bool {
    if len(s) != len(t) {
        return false
    }
    
    var freq [26]int
    
    for idx := 0; idx < len(s); idx++ {
        freq[s[idx] - 'a']++
        freq[t[idx] - 'a']--
    }
    
    for idx := 0; idx < len(freq); idx++ {
        if freq[idx] != 0 {
            return false
        }
    }
    
    return true
}
