# Объяснение

Есть такое понятие в математике, комплимент, это дополнение множества до какого-либо универсума. В рамках этой задачи комплимент это число которое нужно прибавить к i-ому элементу nums, чтобы получить target. Наша задача просто хранить где-то отдельно связку (значение, ключ). Мы смотрим на комплимент, далее ищем его среди ключей мапы и выводим ответ

# Решение

``` python
class Solution:
    def twoSum(self, nums: List[int], target: int) -> List[int]:
        d = dict()

        for i in range(len(nums)):
            complementum = target - nums[i]

            if complementum in d:
                return d[complementum], i

            d[nums[i]] = i
```

``` go 
func twoSum(nums []int, target int) []int {
    answer := make([]int, 2)
    compliment := make(map[int]int)
    for i, num := range nums {
        val, ok := compliment[num]
        if ok {
            answer[0] = val
            answer[1] = i
            break
        } else {
            compliment[target - num] = i
        }
    }

    return answer
}
```
